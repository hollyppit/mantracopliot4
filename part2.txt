async function processUserAnswer(val) {
  hideInput();
  showReplyFlash(val);
  
  const stepInfo = currentStep === 999 
    ? { id: 'verify-mode', question: '작품 검증 Q&A' } 
    : activeSteps[currentStep];

  // 로딩 텍스트
  const txtField = $('dialogTxt');
  const exField = $('dialogExample');
  exField.classList.remove('show');
  txtField.innerHTML = '<span style="color:var(--text-dim);font-weight:normal;font-size:16px;">(정리하고 있습니다...)</span>';
  $('dialogCursor').style.display = 'none';

  // BOARD 태그 추출을 위해 간단 AI 호출 (컨펌 없이)
  const feedback = await fetchAIFeedback(val, stepInfo);

  const boardMatch = feedback.match(/\[BOARD\]:\s*([\s\S]+?)(?:\n|$)/i);
  const boardSummary = boardMatch ? boardMatch[1].trim() : val;
  const visibleFeedback = feedback.replace(/\[BOARD\]:[\s\S]*/i, '').trim();

  if (currentStep !== 999) {
    const stepLabelMap = {
      'l-1':'콘셉트','l-2':'장르','l-3':'로그라인','l-3b':'배경','l-3c':'동기','l-4':'주인공','l-5':'적대자',
      'l-6':'프롤로그','l-7':'문제직면','l-8':'미끼','l-9':'검증','l-10':'신념',
      'l-11':'발전','l-12':'가짜승리','l-13':'방심','l-14':'혼란','l-15':'각성',
      'l-16':'초긴박','l-17':'클라이막스','l-18':'반전','l-19':'이완','l-20':'에필로그',
      's-1':'콘셉트','s-2':'장르','s-3':'로그라인','s-4':'주인공','s-5':'적대자',
      's-6':'코믹릴리프','s-7':'트러블','s-8':'딜레마','s-9':'서스펜스','s-10':'클라이막스','s-11':'해결',
      's-name':'루틴명','s-genre':'대상장르','s-steps':'단계수',
      's-q1':'첫질문','s-q1c':'선택지','s-desc':'소개','s-price':'가격'
    };
    const chipLabel = stepLabelMap[stepInfo.id] || stepInfo.id;
    updateStorySentence(boardSummary, chipLabel);
  }

  // 답변 이력 저장 후 바로 다음 단계 진행
  if (currentStep === 999) {
    // 검증 모드: 짧은 AI 코멘트 보여주고 다시 입력 열기
    typing = false;
    showDialogText({ question: visibleFeedback }, () => {
      showChoices(['계속 질문하기', '검증 종료하고 처음으로'], (c) => {
        hideInput();
        showReplyFlash(c);
        if (c === '검증 종료하고 처음으로') {
          setTimeout(() => goHome(), 1000);
        } else {
          setTimeout(() => {
            showDialogText({ question: `좋습니다. 더 궁금한 점이나 다음 원고 내용을 편하게 적어주세요.` }, () => {
              showInput(false);
            });
          }, 600);
        }
      });
      showInput(false);
    });
  } else {
    // 스토리 빌드업: 확인 없이 즉시 다음 단계
    answerHistory.push({
      stepIdx: currentStep,
      stepId: stepInfo.id,
      value: val,
      boardValue: boardSummary
    });
    // AI 코멘트를 간단히 flash로만 보여주고 바로 다음 단계
    const shortFeedback = visibleFeedback.split('\n')[0].substring(0, 80);
    showReplyFlash(shortFeedback || '✓ 저장됨');
    setTimeout(() => runStep(currentStep + 1), 800);
  }
}

function handleChoice(val) {
  processUserAnswer(val);
}

function handleSend() {
  const box = $('inputBox');
  const val = box.value.trim();
  if (!val) return;
  box.value = ''; box.style.height = '40px';
  processUserAnswer(val);
}

// ── INSIGHT PANEL ──
function hideInput() {
  const area = $('answerArea');
  if (area) area.classList.remove('show');
  const box = $('choicesBox');
  if (box) { box.innerHTML = ''; box.style.display = 'none'; }
}

function showReplyFlash(text) {
  const el = $('replyFlash');
  if (!el) return;
  el.textContent = text.length > 40 ? text.substring(0, 40) + '...' : text;
  el.classList.add('show');
  setTimeout(() => el.classList.remove('show'), 1800);
}

function updateStorySentence(summary, chipLabel) {
  const boxId = isSeller ? 'seller-summary-box' : 'story-summary-box';
  const chipsId = isSeller ? 'seller-chips' : 'story-chips';
  const box = $(boxId);
  const chips = $(chipsId);
  if (box) {
    box.innerHTML = summary;
    box.classList.add('has-content');
    if (isSeller) box.classList.add('seller-content');
  }
  if (chips && chipLabel) {
    const chip = document.createElement('span');
    chip.className = 'sp-chip' + (isSeller ? ' seller-chip' : '');
    chip.textContent = chipLabel;
    chips.appendChild(chip);
  }
}

  root.style.display = 'block';
  
  root.innerHTML = `
    <div style="padding:40px;">
      <div class="board-header" style="margin-bottom:30px;">
        <div class="board-progress-container">
          <div class="board-progress-label">기획 완성도: <span id="boardProgressPercent">0%</span></div>

        </div>

      <div class="board-acts" style="display:flex; flex-direction:column; gap:40px;">
        <div class="board-act">
          <div class="act-label">욕망 (Desire)</div>
          <div class="board-grid" id="actBoard-desire"></div>
        </div>
        <div class="board-act">
          <div class="act-label">갈등 (Conflict)</div>
          <div class="board-grid" id="actBoard-conflict"></div>
        </div>
        <div class="board-act">
          <div class="act-label">극복 (Overcome)</div>
          <div class="board-grid" id="actBoard-overcome"></div>
        </div>
      </div>
      <div class="board-footer" style="margin-top:40px;">
        <button class="board-finish-btn" id="boardFinishBtn" onclick="generateFinalSynopsis()">🪄 최종 시놉시스 생성하기</button>
      </div>
    </div>
  `;

  const acts = {
    desire: $('actBoard-desire'),
    conflict: $('actBoard-conflict'),
    overcome: $('actBoard-overcome')
  };
  
  activeSteps.forEach((step, idx) => {
    let actKey = 'desire';
    if (ACT_MAP.conflict.includes(step.id)) actKey = 'conflict';
    else if (ACT_MAP.overcome.includes(step.id)) actKey = 'overcome';
    
    const container = acts[actKey];
    if (!container) return;
    
    const history = answerHistory.find(h => h.stepId === step.id);
    const card = document.createElement('div');
    card.className = 'board-card' + (history ? ' filled' : '');
    card.onclick = () => openBoardStep(step.id);
    
    card.innerHTML = `
      <div class="board-card-header">
        <div class="board-card-num">STEP ${idx + 1}</div>
        <div class="board-card-check">✓</div>
      </div>
      <div class="board-card-q">${step.question}</div>
      <div class="board-card-a">${history ? (history.boardValue || history.value) : '—'}</div>
    `;
    container.appendChild(card);
  });
  
  updateBoardProgress();

async function saveBoardStep() {
  const val = $('popupInput').value.trim();
  if (!val) { closeBoardPopup(); return; }
  
  const stepInfo = activeSteps.find(s => s.id === currentBoardStepId);
  if (!stepInfo) return;
  
  // Show minimal loading on the button
  const saveBtn = $('popupSaveBtn');
  const originalText = saveBtn.textContent;
  saveBtn.textContent = '정리 중...';
  saveBtn.disabled = true;
  
  try {
    const feedback = await fetchAIFeedback(val, stepInfo);
    const boardMatch = feedback.match(/\[BOARD\]:\s*([\s\S]+?)(?:\n|$)/i);
    const boardSummary = boardMatch ? boardMatch[1].trim() : val;
    
    // Update or Push history
    const existingIdx = answerHistory.findIndex(h => h.stepId === currentBoardStepId);
    const entry = {
      stepIdx: activeSteps.indexOf(stepInfo),
      stepId: stepInfo.id,
      value: val,
      boardValue: boardSummary
    };
    
    if (existingIdx >= 0) answerHistory[existingIdx] = entry;
    else answerHistory.push(entry);
    
    showReplyFlash('✓ 저장되었습니다');
    renderBoard();
    closeBoardPopup();
  } catch (e) {
    showReplyFlash('⚠️ 저장 실패');
  } finally {
    saveBtn.textContent = originalText;
    saveBtn.disabled = false;
  }
}

function initProgressDots(total) {
  const wrap = $('progressDots');

  }

  $('hudProgress').style.opacity = '1';

function updateProgressDots(idx, total) {
  $('stepNum').textContent = idx + 1;
  for (let i = 0; i < total; i++) {
    const d = $('pdot-' + i);
    if (!d) continue;
    d.className = 'pdot' + (i < idx ? ' done' : i === idx ? ' active' : '');
  }

function runStep(idx) {
  closeInsightPanel();

  if (idx >= activeSteps.length) {
    finishFlow();
    return;
  }

  currentStep = idx;
  const step = activeSteps[idx];

  // 필드 업데이트 (스토리보드 패널)
  activeSteps.forEach((s, i) => {
    const el = $(s.fieldId);
    if (el) {
      const filled = answerHistory.find(h => h.stepId === s.id);
      if (filled) {
        el.textContent = filled.value.substring(0, 40);
        el.classList.add(isSeller ? 'seller-filled' : 'filled');
      }
    }
  });

  updateProgressDots(idx, activeSteps.length);
  updateActHud(step.id);

  // 캐릭터 이모지 변경
  const charEmoji = $('charEmoji');
  if (charEmoji && step.visual) {
    charEmoji.classList.add('char-anim-pop');
    setTimeout(() => charEmoji.classList.remove('char-anim-pop'), 600);
    charEmoji.textContent = step.visual;
  }

  // 배너 설정
  if (step.stageIntro) {
    setStageBanner(step.stageIntro);
  } else {
    setStageBanner('');
  }

  // 질문 표시 후 입력창 열기
  showDialogText(step, () => {
    showInput(false);
    // 선택지가 있는 단계면 선택지도 표시
    if (step.choices && step.choices.length) {
      showChoices(step.choices);
    }
  });

  $('backBtn').classList.toggle('show', idx > 0);
  $('homeBtn').classList.add('show');
  $('hudMode').classList.add('show');
}

  const overlay = $('insightOverlay');

  const content = $('insightContent');

  });

let builderSteps = [
  { id: 'step-1', visual: '📝', stageIntro: '이 단계에서는 기본 정보와 콘셉트를 물어봅니다.', question: '가장 먼저 독자에게 묻고 싶은 것은 무엇인가요?', useSolutionBar: true, useDirectInput: true },
  { id: 'step-2', visual: '✨', stageIntro: '핵심 딜레마를 도출하는 단계입니다.', question: '주인공이 겪을 딜레마를 입력해주세요.', useSolutionBar: true, useDirectInput: true }
];

function renderBuilder() {
  const container = $('rbSteps');
  container.innerHTML = '';
  builderSteps.forEach((step, idx) => {
    const card = document.createElement('div');
    card.className = 'rb-step-card';
    card.innerHTML = `
      <div class="rb-step-header">
        <div class="rb-step-title">Step ${idx + 1}</div>
        <div class="rb-step-controls">
          <button class="rb-btn" onclick="moveBuilderStep(${idx}, -1)" ${idx === 0 ? 'disabled style="opacity:0.3"' : ''}>⬆️</button>
          <button class="rb-btn" onclick="moveBuilderStep(${idx}, 1)" ${idx === builderSteps.length - 1 ? 'disabled style="opacity:0.3"' : ''}>⬇️</button>
          <button class="rb-btn delete" onclick="deleteBuilderStep(${idx})">🗑️</button>
        </div>
      </div>
      <div class="rb-field-group">
        <div style="display:flex; gap:12px;">
          <div class="rb-field" style="flex:1;">
            <label>단계 ID (라벨용)</label>
            <input type="text" class="rb-input" value="${step.id || ''}" onchange="updateBuilderStep(${idx}, 'id', this.value)" placeholder="예: 첫인상">
          </div>
          <div class="rb-field" style="width:100px;">
            <label>이모지/시각</label>
            <input type="text" class="rb-input" value="${step.visual || ''}" onchange="updateBuilderStep(${idx}, 'visual', this.value)" placeholder="예: 🚀">
          </div>
        </div>
        <div class="rb-field">
          <label>배경 가이드 (안내 말풍선)</label>
          <textarea class="rb-textarea" onchange="updateBuilderStep(${idx}, 'stageIntro', this.value)" placeholder="이 단계의 목적이나 팁을 작성하세요">${step.stageIntro || ''}</textarea>
        </div>
        <div class="rb-field">
          <label>핵심 질문</label>
          <textarea class="rb-textarea" style="min-height:50px; font-weight:700;" onchange="updateBuilderStep(${idx}, 'question', this.value)" placeholder="사용자에게 던질 결정적 질문을 작성하세요">${step.question || ''}</textarea>
        </div>
        <div class="rb-toggles">
          <label class="rb-toggle-lbl">
            <input type="checkbox" ${step.useSolutionBar !== false ? 'checked' : ''} onchange="updateBuilderStep(${idx}, 'useSolutionBar', this.checked)">
            AI 솔루션 제안 바 표시
          </label>
          <label class="rb-toggle-lbl">
            <input type="checkbox" ${step.useDirectInput !== false ? 'checked' : ''} onchange="updateBuilderStep(${idx}, 'useDirectInput', this.checked)">
            직접 입력 바 표시
          </label>
        </div>
      </div>
    `;
    container.appendChild(card);
  });
}

function updateBuilderStep(idx, field, val) {
  builderSteps[idx][field] = val;
}

function addBuilderStep() {
  builderSteps.push({
    id: 'step-' + (builderSteps.length + 1),
    visual: '🎯',
    stageIntro: '',
    question: '',
    useSolutionBar: true,
    useDirectInput: true
  });
  renderBuilder();
}

function deleteBuilderStep(idx) {
  if (confirm('이 단계를 삭제하시겠습니까?')) {
    builderSteps.splice(idx, 1);
    renderBuilder();
  }
}

function moveBuilderStep(idx, dir) {
  if (idx + dir < 0 || idx + dir >= builderSteps.length) return;
  const temp = builderSteps[idx];
  builderSteps[idx] = builderSteps[idx + dir];
  builderSteps[idx + dir] = temp;
  renderBuilder();
}

  activeSteps = builderSteps;
  sellerSteps = builderSteps; // Override sellerSteps so it saves
  initProgressDots(activeSteps.length);
  $('routineBuilder').style.display = 'none';
  document.querySelector('.dialog-area').style.display = 'flex';
  runStep(0);
}

async function getCurrentUser() {
  if (currentUser) return currentUser;
  const { data: { user } } = await _sb.auth.getUser();

function getCurrentProjectId() {
  let pid = sessionStorage.getItem('mantra_current_pid');
  if (!pid) {
    pid = 'p_' + Date.now() + '_' + Math.random().toString(36).slice(2, 8);
    sessionStorage.setItem('mantra_current_pid', pid);
  }
  return pid;
}

async function saveProjectToStorage(synopsis) {
  const user = await getCurrentUser();
  const guestId = localStorage.getItem('mantra_guest_id') || (() => {
    const id = Date.now().toString(36);
    localStorage.setItem('mantra_guest_id', id);
    return id;
  })();
  const uid = user ? user.id : `guest_${guestId}`;
  const pid = getCurrentProjectId();

  const titleAnswer = answerHistory.find(h => ['l-3','s-3','l-1','s-1'].includes(h.stepId));
  const title = titleAnswer
    ? (titleAnswer.value.length > 30 ? titleAnswer.value.slice(0, 30) + '...' : titleAnswer.value)
    : '제목 없는 작품';
  const genreAnswer = answerHistory.find(h => h.stepId === 'l-2' || h.stepId === 's-2');
  const genre = genreAnswer ? genreAnswer.value.slice(0, 20) : '';

  const project = {
    id: pid, uid, title, genre,
    synopsis: synopsis || '',
    routineType: activeSteps === shortSteps ? 'short' : 'long',
    copilotName: selectedCharName,
    answerHistory,
    completedAt: new Date().toISOString(),
    updatedAt: new Date().toISOString(),
  };

  // localStorage 즉시 저장
  localStorage.setItem(`mantra_project_${pid}`, JSON.stringify(project));
  const idx = JSON.parse(localStorage.getItem(`mantra_index_${uid}`) || '[]');
  const ei = idx.findIndex(p => p.id === pid);
  const meta = { id: pid, title, genre, completedAt: project.completedAt };
  if (ei >= 0) idx[ei] = meta; else idx.unshift(meta);
  localStorage.setItem(`mantra_index_${uid}`, JSON.stringify(idx));

  // Supabase 저장 (로그인 시)
  if (user) {
    try {
      const sb = getSb();
      await sb.from('projects').upsert({
        id: pid, user_id: uid, title, genre,
        synopsis: synopsis || '',
        routine_type: project.routineType,
        copilot_name: selectedCharName,
        answer_history: answerHistory,
        completed_at: project.completedAt,
        updated_at: project.updatedAt,
      });
    } catch (e) { console.warn('Supabase 저장 실패:', e); }
  }

  showReplyFlash('💾 작품이 저장됐습니다');
}
