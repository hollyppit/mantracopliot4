// ── CONTEXT CHAT ──
async function handleContextSend() {
  const box = $('contextInputBox');
  const userText = box.value.trim();
  if (!userText) return;
  
  box.value = '';
  appendChatMessage('user', userText);
  
  const loadingId = appendChatMessage('assistant', '(생각 중...)');
  
  try {
    const systemPrompt = getSystemPromptForRoutine();
    const response = await fetch('/api/chat', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({
        messages: [
          { role: 'system', content: systemPrompt },
          { role: 'user', content: userText }
        ],
        projectContext: {
          title: selectedProject.title,
          genre: selectedProject.genre,
          synopsis: selectedProject.synopsis,
          answerHistory: selectedProject.answer_history
        }
      })
    });
    
    const data = await response.json();
    const aiText = data.choices?.[0]?.message?.content || '죄송합니다. 오류가 발생했습니다.';
    updateChatMessage(loadingId, aiText);
  } catch (e) {
    updateChatMessage(loadingId, '네트워크 오류가 발생했습니다.');
  }
}

  if (currentRoutine === 'foreshadow') return "당신은 치밀한 서사 검토자입니다. 작품 전체의 복선과 설정을 대조하여 모순되는 부분을 지적하고, 서사적 완결성을 높일 수 있는 디테일을 제안하세요.";

}
